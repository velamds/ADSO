🏗️ Creacionales

Creacional

Singleton

Garantiza que una clase tenga una única instancia y proporciona un punto global de acceso a ella.

¿Cuándo usarlo?

Cuando exactamente un objeto debe coordinar acciones en todo el sistema: configuración, conexión a base de datos, logging, caché.

Producto real

Spotify / Netflix: El módulo de configuración carga variables de entorno una sola vez al arrancar y todas las partes de la app leen de la misma instancia.

AppConfig — Spotify-like config manager 
<?php

class AppConfig {
    private static ?AppConfig $instance = null;
    private array $data = [];
    private bool  $loaded = false;

    private function __construct() {}
    private function __clone()    {}

    public static function getInstance(): static {
        if (static::$instance === null) {
            static::$instance = new static();
        }
        return static::$instance;
    }

    public function load(string $env = 'production'): void {
        if ($this->loaded) return;
        $this->data = [
            'db_url'      => "postgresql://db-{$env}.internal:5432/spotify",
            'redis_url'   => 'redis://cache.internal:6379',
            'cache_ttl'   => 3600,
            'max_streams' => 5,
            'feature_flags' => [
                'new_home_feed' => true,
                'podcast_v2'   => false,
                'hifi_audio'   => $env === 'production',
            ],
        ];
        $this->loaded = true;
    }

    public function get(string $key, mixed $default = null): mixed {
        return $this->data[$key] ?? $default;
    }

    public function feature(string $flag): bool {
        return $this->data['feature_flags'][$flag] ?? false;
    }
}

// ── Uso ──────────────────────────────────────────────────────
$configA = AppConfig::getInstance();
$configA->load();
echo $configA->get('max_streams');          // 5

$configB = AppConfig::getInstance();        // misma instancia
echo $configB->get('cache_ttl');            // 3600

var_dump($configA === $configB);            // bool(true) ← MISMA instancia

if (AppConfig::getInstance()->feature('hifi_audio')) {
    echo 'Activando audio HiFi 24-bit...';
}
public sealed class AppConfig
{
    private static AppConfig? _instance;
    private static readonly object _lock = new();
    private Dictionary<string, object> _data = new();
    private bool _loaded;

    private AppConfig() { }

    public static AppConfig Instance
    {
        get {
            if (_instance is null)
                lock (_lock) { _instance ??= new AppConfig(); }
            return _instance;
        }
    }

    public void Load(string env = "production")
    {
        if (_loaded) return;
        _data = new()
        {
            ["db_url"]      = $"postgresql://db-{env}.internal:5432/spotify",
            ["redis_url"]   = "redis://cache.internal:6379",
            ["cache_ttl"]   = 3600,
            ["max_streams"] = 5,
            ["feature_flags"] = new Dictionary<string, bool>
            {
                ["new_home_feed"] = true,
                ["podcast_v2"]   = false,
                ["hifi_audio"]   = env == "production",
            },
        };
        _loaded = true;
    }

    public T? Get<T>(string key, T? def = default) =>
        _data.TryGetValue(key, out var v) ? (T)v : def;

    public bool Feature(string flag)
    {
        var flags = Get<Dictionary<string, bool>>("feature_flags");
        return flags?.GetValueOrDefault(flag) ?? false;
    }
}

// ── Uso ──────────────────────────────────────────────────────
var configA = AppConfig.Instance;
configA.Load();
Console.WriteLine(configA.Get<int>("max_streams"));   // 5

var configB = AppConfig.Instance;                      // misma instancia
Console.WriteLine(configB.Get<int>("cache_ttl"));     // 3600

Console.WriteLine(ReferenceEquals(configA, configB)); // True ← MISMA instancia

if (AppConfig.Instance.Feature("hifi_audio"))
    Console.WriteLine("Activando audio HiFi 24-bit...");

Creacional

Factory Method

Define una interfaz para crear un objeto, pero deja que las subclases decidan qué clase instanciar.

¿Cuándo usarlo?

Cuando el código necesita trabajar con distintas variantes de un objeto sin acoplarse a sus implementaciones concretas.

Producto real

Shopify / MercadoPago: El checkout soporta múltiples métodos de pago. La lógica de negocio no conoce (ni le importa) si el cobro es por Stripe, PayPal o criptomonedas.

payments — Pasarela de pagos multi-proveedor 
<?php

abstract class PaymentProcessor {
    abstract public function charge(float $amount, string $currency, string $token): array;
    abstract public function refund(string $transactionId): bool;

    protected function fmt(float $amount, string $currency): string {
        return "{$currency} " . number_format($amount, 2);
    }
}

class StripeProcessor extends PaymentProcessor {
    public function charge(float $amount, string $currency, string $token): array {
        echo "[Stripe] Cobrando {$this->fmt($amount, $currency)}\n";
        return ['id' => 'ch_3Nxy...', 'status' => 'succeeded', 'provider' => 'stripe'];
    }
    public function refund(string $transactionId): bool {
        echo "[Stripe] Reembolsando {$transactionId}\n";
        return true;
    }
}

class PayPalProcessor extends PaymentProcessor {
    public function charge(float $amount, string $currency, string $token): array {
        echo "[PayPal] Cobrando {$this->fmt($amount, $currency)}\n";
        return ['id' => 'PAYID-xxx', 'status' => 'COMPLETED', 'provider' => 'paypal'];
    }
    public function refund(string $transactionId): bool {
        echo "[PayPal] Reembolsando {$transactionId}\n";
        return true;
    }
}

class CryptoProcessor extends PaymentProcessor {
    public function charge(float $amount, string $currency, string $token): array {
        echo "[Crypto] Registrando tx de {$this->fmt($amount, $currency)}\n";
        return ['tx_hash' => '0xabc...', 'status' => 'confirmed', 'provider' => 'crypto'];
    }
    public function refund(string $transactionId): bool {
        echo "[Crypto] Las transacciones cripto son irreversibles.\n";
        return false;
    }
}

// ── Factory ──────────────────────────────────────────────────
function createProcessor(string $method): PaymentProcessor {
    return match($method) {
        'stripe' => new StripeProcessor(),
        'paypal' => new PayPalProcessor(),
        'crypto' => new CryptoProcessor(),
        default  => throw new InvalidArgumentException("Método desconocido: '{$method}'"),
    };
}

// ── Lógica de negocio (no conoce las clases concretas) ───────
function processCheckout(float $total, string $method, string $token): array {
    $processor = createProcessor($method);
    $result    = $processor->charge($total, 'MXN', $token);
    echo "Orden confirmada. ID: {$result['id']}\n";
    return $result;
}

Creacional

Builder

Separa la construcción de un objeto complejo de su representación para que el mismo proceso pueda crear distintas representaciones.

¿Cuándo usarlo?

Cuando construir un objeto requiere muchos pasos opcionales o cuando el constructor tendría demasiados parámetros.

Producto real

Django ORM / SQLAlchemy: La API .filter().order_by().limit() es un Builder que construye una consulta SQL compleja de forma legible y encadenada.

QueryBuilder — ORM query builder (estilo Laravel/Eloquent) 
<?php

class QueryBuilder {
    private string  $table;
    private array   $cols   = ['*'];
    private array   $joins  = [];
    private array   $wheres = [];
    private ?string $order  = null;
    private ?int    $lim    = null;

    public function __construct(string $table) {
        $this->table = $table;
    }

    // Cada método retorna $this para encadenamiento fluido
    public function select(string ...$cols): static {
        $this->cols = $cols; return $this;
    }
    public function join(string $table, string $on): static {
        $this->joins[] = "JOIN {$table} ON {$on}"; return $this;
    }
    public function where(string $cond): static {
        $this->wheres[] = $cond; return $this;
    }
    public function orderBy(string $col, string $dir = 'ASC'): static {
        $this->order = "{$col} {$dir}"; return $this;
    }
    public function limit(int $n): static {
        $this->lim = $n; return $this;
    }

    public function build(): string {
        $sql = 'SELECT ' . implode(', ', $this->cols) . " FROM {$this->table}";
        foreach ($this->joins as $j)  { $sql .= "\n  {$j}"; }
        if ($this->wheres) { $sql .= "\n  WHERE " . implode(' AND ', $this->wheres); }
        if ($this->order)  { $sql .= "\n  ORDER BY {$this->order}"; }
        if ($this->lim)    { $sql .= "\n  LIMIT {$this->lim}"; }
        return $sql;
    }
}

public class QueryBuilder
{
    private readonly string _table;
    private List<string> _cols   = ["*"];
    private List<string> _joins  = [];
    private List<string> _wheres = [];
    private string? _order;
    private int?    _lim;

    public QueryBuilder(string table) => _table = table;

    // Cada método retorna this para encadenamiento fluido
    public QueryBuilder Select(params string[] cols)
        { _cols = [.. cols]; return this; }
    public QueryBuilder Join(string table, string on)
        { _joins.Add($"JOIN {table} ON {on}"); return this; }
    public QueryBuilder Where(string cond)
        { _wheres.Add(cond); return this; }
    public QueryBuilder OrderBy(string col, string dir = "ASC")
        { _order = $"{col} {dir}"; return this; }
    public QueryBuilder Limit(int n)
        { _lim = n; return this; }

    public string Build()
    {
        var sql = $"SELECT {string.Join(", ", _cols)} FROM {_table}";
        foreach (var j in _joins)  sql += $"\n  {j}";
        if (_wheres.Count > 0)     sql += $"\n  WHERE {string.Join(" AND ", _wheres)}";
        if (_order is not null)    sql += $"\n  ORDER BY {_order}";
        if (_lim.HasValue)         sql += $"\n  LIMIT {_lim}";
        return sql;
    }
}

// ── Consulta 1: panel de órdenes pendientes ──────────────────
var ordersQuery = new QueryBuilder("orders")
    .Select("orders.id", "users.email", "orders.total", "orders.status")
    .Join("users", "orders.user_id = users.id")

🔧 Estructurales

Estructural

Adapter

Convierte la interfaz de una clase en otra que el cliente espera. Permite que trabajen juntas clases con interfaces incompatibles.

¿Cuándo usarlo?

Cuando necesitas usar una biblioteca o API externa que tiene una interfaz diferente a la que tu sistema espera, sin modificar ninguno de los dos.

Producto real

Uber / Rappi: La app integra Google Maps, OpenStreetMap y HERE Maps. Un Adapter por proveedor permite cambiar de SDK sin tocar la lógica de enrutamiento.

maps — Integración de SDKs de mapas (Uber/Rappi) 
<?php

// ── Interfaz que NUESTRO sistema espera ──────────────────────
interface MapService {
    public function getRoute(array $origin, array $destination): array; // [lat, lng]
    public function geocode(string $address): array;                    // [lat, lng]
}

// ── SDK externo de Google Maps (no podemos modificarlo) ──────
class GoogleMapsSDK {
    public function getDirections(float $oLat, float $oLng, float $dLat, float $dLng): array {
        return ['distance_m' => 5200, 'duration_s' => 720, 'status' => 'OK'];
    }
    public function geocodeAddress(string $address): array {
        return ['lat' => 19.4326, 'lng' => -99.1332];
    }
}

// ── Adapter: traduce nuestra interfaz a la del SDK ───────────
class GoogleMapsAdapter implements MapService {
    private GoogleMapsSDK $sdk;

    public function __construct() { $this->sdk = new GoogleMapsSDK(); }

    public function getRoute(array $origin, array $destination): array {
        $raw = $this->sdk->getDirections($origin[0], $origin[1], $destination[0], $destination[1]);
        return [
            'distance_km'  => $raw['distance_m'] / 1000,
            'duration_min' => intdiv($raw['duration_s'], 60),
        ];
    }

    public function geocode(string $address): array {
        $raw = $this->sdk->geocodeAddress($address);
        return [$raw['lat'], $raw['lng']];
    }
}

// ── Lógica de negocio: solo conoce MapService ────────────────
function calculateDelivery(MapService $maps, string $pickup, string $dropoff): array {
    $origin = $maps->geocode($pickup);
    $dest   = $maps->geocode($dropoff);
    $route  = $maps->getRoute($origin, $dest);
    return [...$route, 'fare' => $route['distance_km'] * 8.5];
}

// Cambiamos proveedor sin tocar la lógica de negocio
$delivery = calculateDelivery(new GoogleMapsAdapter(), 'Reforma 222, CDMX', 'Polanco, CDMX');
printf("Distancia: %.1f km | Tarifa: $%.2f\n", $delivery['distance_km'], $delivery['fare']);
// ── Interfaz que NUESTRO sistema espera ──────────────────────
public interface IMapService
{
    RouteResult GetRoute((double Lat, double Lng) origin, (double Lat, double Lng) dest);
    (double Lat, double Lng) Geocode(string address);
}

public record RouteResult(double DistanceKm, int DurationMin);

// ── SDK externo de Google Maps (no podemos modificarlo) ──────
public class GoogleMapsSDK
{
    public (int DistanceM, int DurationS) GetDirections(
        double oLat, double oLng, double dLat, double dLng) => (5200, 720);

    public (double Lat, double Lng) GeocodeAddress(string address) => (19.4326, -99.1332);
}

// ── Adapter: traduce nuestra interfaz a la del SDK ───────────
public class GoogleMapsAdapter : IMapService
{
    private readonly GoogleMapsSDK _sdk = new();

    public RouteResult GetRoute((double Lat, double Lng) origin, (double Lat, double Lng) dest)
    {
        var (distM, durS) = _sdk.GetDirections(origin.Lat, origin.Lng, dest.Lat, dest.Lng);
        return new(DistanceKm: distM / 1000.0, DurationMin: durS / 60);
    }

    public (double Lat, double Lng) Geocode(string address) =>
        _sdk.GeocodeAddress(address);
}

// ── Lógica de negocio: solo conoce IMapService ───────────────
static (double DistanceKm, int DurationMin, double Fare) CalculateDelivery(
    IMapService maps, string pickup, string dropoff)
{
    var origin = maps.Geocode(pickup);
    var dest   = maps.Geocode(dropoff);
    var route  = maps.GetRoute(origin, dest);
    return (route.DistanceKm, route.DurationMin, route.DistanceKm * 8.5);
}

Console.WriteLine($"Distancia: {d.DistanceKm} km | Tarifa: ${d.Fare:F2}");

Estructural

Decorator

Agrega responsabilidades adicionales a un objeto de forma dinámica, como una alternativa flexible a la herencia.

¿Cuándo usarlo?

Cuando necesitas agregar comportamiento a funciones o clases de manera composable y sin modificarlas: logging, autenticación, caché, rate limiting.

Producto real

Express.js / Django / FastAPI: El middleware es puro Decorator. Cada capa (auth, logging, rate limit) envuelve al handler sin que éste sepa nada de ellas.

middleware — Middleware API (estilo Laravel/ASP.NET) 
<?php

// ── Interfaz del handler ──────────────────────────────────────
interface RequestHandler {
    public function handle(array $request): array;
}

// ── Handler base (núcleo, sin saber nada de las capas) ───────
class DashboardHandler implements RequestHandler {

class RateLimitDecorator extends HandlerDecorator {
    private array $log = [];
    public function __construct(RequestHandler $inner, private int $maxPerMin = 60) {
        parent::__construct($inner);
    }
    public function handle(array $request): array {
        $user = $request['user_id'] ?? $request['ip'] ?? 'anon';
        $now  = time();
        $this->log[$user] = array_filter($this->log[$user] ?? [], fn($t) => $now - $t < 60);
        if (count($this->log[$user]) >= $this->maxPerMin) {
            return ['error' => 'Rate limit excedido', 'status' => 429];
        }
        $this->log[$user][] = $now;
        return $this->inner->handle($request);
    }
}

public abstract class HandlerDecorator(IRequestHandler inner) : IRequestHandler
{
    protected readonly IRequestHandler _inner = inner;
    public abstract Dictionary<string, object> Handle(Dictionary<string, object> request);
}

public class LogDecorator(IRequestHandler inner) : HandlerDecorator(inner)
{
    public override Dictionary<string, object> Handle(Dictionary<string, object> request)
    {
        Console.WriteLine($"[LOG]  {request["method"]} {request["path"]}");
        var response = _inner.Handle(request);
        Console.WriteLine($"[LOG]  Status: {response["status"]}");
        return response;
    }
}

public class AuthDecorator(IRequestHandler inner) : HandlerDecorator(inner)
{
    public override Dictionary<string, object> Handle(Dictionary<string, object> request)
    {
        if (request.GetValueOrDefault("token") is null)
            return new() { ["error"] = "No autorizado", ["status"] = 401 };
        request["user_id"] = 42; // validar JWT en producción
        return _inner.Handle(request);
    }
}

public class RateLimitDecorator(IRequestHandler inner, int maxPerMin = 60) : HandlerDecorator(inner)
{
    private readonly Dictionary<string, List<long>> _log = new();
    public override Dictionary<string, object> Handle(Dictionary<string, object> request)
    {
        var user = request.GetValueOrDefault("user_id")?.ToString() ?? "anon";
        var now  = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
        _log.TryAdd(user, []);
        _log[user].RemoveAll(t => now - t >= 60);
        if (_log[user].Count >= maxPerMin)
            return new() { ["error"] = "Rate limit excedido", ["status"] = 429 };
        _log[user].Add(now);
        return _inner.Handle(request);
    }
}

Comportamental

Observer

Define una dependencia uno-a-muchos: cuando un objeto cambia de estado, todos sus dependientes son notificados automáticamente.

¿Cuándo usarlo?

Cuando un cambio en un objeto requiere actualizar otros, y no sabes cuántos son ni quiénes son. Eventos de dominio, sistemas reactivos, webhooks.

Producto real

GitHub Actions / Slack: Cuando un build falla, el sistema publica un evento. Email, Slack, PagerDuty y audit log reaccionan de forma independiente.

Strategy

Define una familia de algoritmos, los encapsula y los hace intercambiables en tiempo de ejecución.

¿Cuándo usarlo?

Cuando tienes múltiples variantes de un algoritmo y necesitas intercambiarlos según condiciones de runtime (tipo de usuario, plan, contexto).

Producto real

Comportamental

Command

Encapsula una solicitud como un objeto, lo que permite parametrizar clientes, encolar solicitudes y soportar operaciones reversibles.

¿Cuándo usarlo?

Cuando necesitas historial de operaciones, deshacer/rehacer, colas de trabajo, transacciones o auditoría. También útil para macro-comandos.

Producto real

Google Docs / Figma / VS Code: Cada acción del usuario (escribir, formatear, mover) es un Command. El historial de Commands permite Ctrl+Z y Ctrl+Y ilimitados.

editor — Editor con undo/redo (estilo Google Docs / Figma) 
<?php

// ── Estado del documento ──────────────────────────────────────
class Document {
    public string $content  = '';
    public int    $fontSize = 12;
    public bool   $isBold   = false;
}

// ── Interfaz Command ──────────────────────────────────────────
interface Command {
    public function execute(): void;
    public function undo(): void;
    public function description(): string;
}

// ── Comandos concretos ────────────────────────────────────────
class InsertTextCommand implements Command {
    private int $pos;

    public function __construct(private Document $doc, private string $text, ?int $pos = null) {
        $this->pos = $pos ?? strlen($doc->content);
    }

    public function execute(): void {
        $this->doc->content = substr($this->doc->content, 0, $this->pos)
            . $this->text . substr($this->doc->content, $this->pos);
    }
    public function undo(): void {
        $this->doc->content = substr($this->doc->content, 0, $this->pos)
            . substr($this->doc->content, $this->pos + strlen($this->text));
    }
    public function description(): string { return "InsertText(\"{$this->text}\")"; }
}

class FormatCommand implements Command {
    private mixed $oldValue;

    public function __construct(
        private Document $doc, private string $prop, private mixed $newValue
    ) {
        $this->oldValue = $doc->{$prop};
    }

    public function execute(): void { $this->doc->{$this->prop} = $this->newValue; }
    public function undo():    void { $this->doc->{$this->prop} = $this->oldValue; }
    public function description(): string {
        return "Format({$this->prop}: {$this->oldValue} → {$this->newValue})";
    }
}

// ── Historial de comandos ─────────────────────────────────────
class CommandHistory {
    private array $done = [], $undone = [];

    public function execute(Command $cmd): void {
        $cmd->execute();
        $this->done[]  = $cmd;
        $this->undone  = []; // nueva acción borra el redo stack
        echo "  DO:   " . $cmd->description() . "\n";
    }
    public function undo(): void {
        if (!$this->done) { echo "Nada que deshacer.\n"; return; }
        $cmd = array_pop($this->done);
        $cmd->undo();
        $this->undone[] = $cmd;
        echo "  UNDO: " . $cmd->description() . "\n";
    }
    public function redo(): void {
        if (!$this->undone) { echo "Nada que rehacer.\n"; return; }
        $cmd = array_pop($this->undone);
        $cmd->execute();
        $this->done[] = $cmd;
        echo "  REDO: " . $cmd->description() . "\n";
    }
}

public class Document
{
    public string Content  { get; set; } = "";
    public int    FontSize { get; set; } = 12;
    public bool   IsBold   { get; set; } = false;
}

// ── Interfaz Command ──────────────────────────────────────────
public interface ICommand
{
    void   Execute();
    void   Undo();
    string Description { get; }
}

// ── Comandos concretos ────────────────────────────────────────
public class InsertTextCommand(Document doc, string text, int? pos = null) : ICommand
{
    private readonly int _pos = pos ?? doc.Content.Length;

    public void Execute() =>
        doc.Content = doc.Content[.._pos] + text + doc.Content[_pos..];
    public void Undo() =>
        doc.Content = doc.Content[.._pos] + doc.Content[(_pos + text.Length)..];
    public string Description => $"InsertText(\"{text}\")";
}

public class FormatCommand<T>(
    Document doc, Func<Document, T> get, Action<Document, T> set, T newVal) : ICommand
{
    private readonly T _old = get(doc);
    public void Execute() => set(doc, newVal);
    public void Undo()    => set(doc, _old);
    public string Description => $"Format({_old} → {newVal})";
}

// ── Historial de comandos ─────────────────────────────────────
public class CommandHistory
{
    private readonly Stack<ICommand> _done   = new();
    private readonly Stack<ICommand> _undone = new();

    public void Execute(ICommand cmd)
    {
        cmd.Execute(); _done.Push(cmd); _undone.Clear();
        Console.WriteLine($"  DO:   {cmd.Description}");
    }
    public void Undo()
    {
        if (_done.Count == 0) { Console.WriteLine("Nada que deshacer."); return; }
        var cmd = _done.Pop(); cmd.Undo(); _undone.Push(cmd);
        Console.WriteLine($"  UNDO: {cmd.Description}");
    }
    public void Redo()
    {
        if (_undone.Count == 0) { Console.WriteLine("Nada que rehacer."); return; }
        var cmd = _undone.Pop(); cmd.Execute(); _done.Push(cmd);
        Console.WriteLine($"  REDO: {cmd.Description}");
    }
}

// ── Simulación de sesión de edición ──────────────────────────
var doc     = new Document();
var history = new CommandHistory();

history.Execute(new InsertTextCommand(doc, "Hola, mundo!"));
history.Execute(new FormatCommand<int>(doc,  d => d.FontSize, (d, v) => d.FontSize = v, 24));
history.Execute(new FormatCommand<bool>(doc, d => d.IsBold,   (d, v) => d.IsBold   = v, true));

Console.WriteLine($"\nDoc: '{doc.Content}' | size={doc.FontSize} | bold={doc.IsBold}\n");

history.Undo(); // deshace IsBold
history.Undo(); // deshace FontSize
history.Redo(); // rehace FontSize
