1 Instalación en Windows

Descarga e instala los servidores. Ambos pueden coexistir en la misma máquina.

  1. Descarga MySQL Installer desde dev.mysql.com/downloads/installer
  2. Elige Developer Default e instala MySQL Server + MySQL Shell
  3. Durante la configuración, elige puerto 3306 y establece la contraseña del usuario root
  4. Al finalizar, MySQL se registra como servicio de Windows y arranca automáticamente
  5. Agrega al PATH: C:\Program Files\MySQL\MySQL Server 8.x\bin
Verificar desde CMD: Abre CMD o PowerShell y ejecuta: mysql --version
CMD — verificar instalación
-- En CMD de Windows:
mysql --version
-- Salida esperada:
-- mysql  Ver 8.x.x Distrib 8.x.x, for Win64 (x86_64)
  1. Descarga el instalador de EDB desde postgresql.org/download/windows
  2. Instala PostgreSQL Server + pgAdmin 4 + herramientas de línea de comandos
  3. Puerto por defecto: 5432. El superusuario se llama postgres
  4. Al finalizar, el servicio se inicia automáticamente en Windows
  5. Agrega al PATH: C:\Program Files\PostgreSQL\16\bin
Verificar desde CMD: Abre CMD o PowerShell y ejecuta: psql --version
CMD — verificar instalación
-- En CMD de Windows:
psql --version
-- Salida esperada:
-- psql (PostgreSQL) 16.x

2 Conectarse al servidor

Siempre desde CMD o PowerShell como administrador.

MySQL — conectar
-- Conectar como root al servidor local:
mysql -u root -p

-- Conectar a un servidor remoto:
mysql -h 192.168.1.100 -P 3306 -u root -p

-- Conectar y seleccionar base de datos directamente:
mysql -u root -p mi_base_datos

-- Salir del cliente:
EXIT;
-- o también: QUIT;  o  \q
Comandos útiles dentro del cliente MySQL: \s — estado del servidor · \h — ayuda · SHOW DATABASES; · USE nombre;
PostgreSQL — conectar
-- Conectar como postgres al servidor local:
psql -U postgres

-- Conectar a una base de datos específica:
psql -U postgres -d mi_base_datos

-- Conectar a servidor remoto:
psql -h 192.168.1.100 -p 5432 -U postgres -d mi_base_datos

-- Salir del cliente:
\q
Meta-comandos útiles en psql: \l — listar bases · \dt — listar tablas · \d tabla — describir tabla · \? — ayuda

4 Bases de datos

Crear, listar, seleccionar y eliminar bases de datos.

MySQL
-- Crear base de datos con codificación UTF-8
CREATE DATABASE tienda
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

-- Listar todas las bases de datos
SHOW DATABASES;

-- Seleccionar / usar una base de datos
USE tienda;

-- Ver la base de datos activa
SELECT DATABASE();

-- Eliminar base de datos (¡irreversible!)
DROP DATABASE IF EXISTS tienda;
PostgreSQL
-- Crear base de datos
CREATE DATABASE tienda
  ENCODING 'UTF8'
  LC_COLLATE 'es_ES.UTF-8'
  TEMPLATE template0;

-- Listar bases de datos (meta-comando psql)
\l

-- O con SQL:
SELECT datname FROM pg_database;

-- Conectarse a otra base de datos (meta-comando)
\c tienda

-- Ver la base de datos activa
SELECT current_database();

-- Eliminar base de datos (¡irreversible!)
DROP DATABASE IF EXISTS tienda;

5 Crear tablas

La estructura de la tabla define las columnas, tipos y restricciones.

MySQL
-- Tabla de clientes
CREATE TABLE clientes (
  id          INT AUTO_INCREMENT PRIMARY KEY,
  nombre      VARCHAR(100) NOT NULL,
  email       VARCHAR(150) UNIQUE NOT NULL,
  telefono    VARCHAR(20),
  activo      TINYINT(1) DEFAULT 1,
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Tabla de productos
CREATE TABLE productos (
  id          INT AUTO_INCREMENT PRIMARY KEY,
  nombre      VARCHAR(200) NOT NULL,
  precio      DECIMAL(10,2) NOT NULL,
  stock       INT DEFAULT 0,
  categoria   VARCHAR(80)
);

-- Ver las tablas de la base activa
SHOW TABLES;

-- Ver estructura de una tabla
DESCRIBE clientes;

-- Modificar tabla: agregar columna
ALTER TABLE clientes ADD COLUMN ciudad VARCHAR(80);

-- Eliminar tabla
DROP TABLE IF EXISTS productos;
PostgreSQL
-- Tabla de clientes
CREATE TABLE clientes (
  id          SERIAL PRIMARY KEY,
  nombre      VARCHAR(100) NOT NULL,
  email       VARCHAR(150) UNIQUE NOT NULL,
  telefono    VARCHAR(20),
  activo      BOOLEAN DEFAULT TRUE,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Tabla de productos
CREATE TABLE productos (
  id          SERIAL PRIMARY KEY,
  nombre      VARCHAR(200) NOT NULL,
  precio      NUMERIC(10,2) NOT NULL,
  stock       INTEGER DEFAULT 0,
  categoria   VARCHAR(80)
);

-- Ver tablas en el esquema actual
\dt

-- Ver estructura de una tabla
\d clientes

-- Modificar tabla: agregar columna
ALTER TABLE clientes ADD COLUMN ciudad VARCHAR(80);

-- Eliminar tabla
DROP TABLE IF EXISTS productos;

6 Restricciones (Constraints)

Las restricciones garantizan la integridad de los datos.

MySQL — Tabla con todas las restricciones
CREATE TABLE pedidos (
  id            INT AUTO_INCREMENT,
  cliente_id    INT NOT NULL,
  producto_id   INT NOT NULL,
  cantidad      INT NOT NULL,
  total         DECIMAL(10,2) NOT NULL,
  estado        ENUM('pendiente','pagado','cancelado') DEFAULT 'pendiente',
  created_at    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  -- Restricciones
  PRIMARY KEY (id),
  CONSTRAINT fk_pedido_cliente
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
    ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_pedido_producto
    FOREIGN KEY (producto_id) REFERENCES productos(id),
  CONSTRAINT chk_cantidad CHECK (cantidad > 0),
  CONSTRAINT chk_total CHECK (total >= 0)
) ENGINE=InnoDB;
PostgreSQL — Tabla con todas las restricciones
-- Primero crear el tipo ENUM
CREATE TYPE estado_pedido AS ENUM ('pendiente','pagado','cancelado');

CREATE TABLE pedidos (
  id            SERIAL,
  cliente_id    INTEGER NOT NULL,
  producto_id   INTEGER NOT NULL,
  cantidad      INTEGER NOT NULL,
  total         NUMERIC(10,2) NOT NULL,
  estado        estado_pedido DEFAULT 'pendiente',
  created_at    TIMESTAMPTZ DEFAULT NOW(),

  -- Restricciones
  PRIMARY KEY (id),
  CONSTRAINT fk_pedido_cliente
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
    ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT fk_pedido_producto
    FOREIGN KEY (producto_id) REFERENCES productos(id),
  CONSTRAINT chk_cantidad CHECK (cantidad > 0),
  CONSTRAINT chk_total CHECK (total >= 0)
);

8 SELECT — Consultar datos

MySQL
-- Seleccionar todas las columnas
SELECT * FROM clientes;

-- Seleccionar columnas específicas
SELECT nombre, email FROM clientes;

-- Usar alias para columnas
SELECT
  nombre  AS cliente,
  email   AS correo,
  activo  AS habilitado
FROM clientes;

-- Ordenar resultados
SELECT * FROM productos
ORDER BY precio DESC;

-- Limitar resultados
SELECT * FROM productos
ORDER BY precio DESC LIMIT 5;

-- Paginación con LIMIT + OFFSET
SELECT * FROM clientes
LIMIT 10 OFFSET 20;  -- página 3 de 10 por página

-- Valores distintos
SELECT DISTINCT categoria FROM productos;
PostgreSQL
-- Seleccionar todas las columnas
SELECT * FROM clientes;

-- Seleccionar columnas específicas
SELECT nombre, email FROM clientes;

-- Usar alias para columnas
SELECT
  nombre  AS cliente,
  email   AS correo,
  activo  AS habilitado
FROM clientes;

-- Ordenar resultados
SELECT * FROM productos
ORDER BY precio DESC;

-- Limitar resultados
SELECT * FROM productos
ORDER BY precio DESC LIMIT 5;

-- Paginación con LIMIT + OFFSET
SELECT * FROM clientes
LIMIT 10 OFFSET 20;

-- Valores distintos
SELECT DISTINCT categoria FROM productos;

-- FETCH FIRST (estándar SQL, equivalente a LIMIT)
SELECT * FROM productos
ORDER BY precio DESC
FETCH FIRST 5 ROWS ONLY;

9 WHERE — Filtrar datos

MySQL
-- Comparaciones básicas
SELECT * FROM productos WHERE precio > 100;
SELECT * FROM clientes  WHERE activo = 1;

-- Operadores lógicos AND / OR / NOT
SELECT * FROM productos
WHERE precio > 50 AND stock > 0;

SELECT * FROM productos
WHERE categoria = 'Electrónica' OR categoria = 'Periféricos';

-- IN: lista de valores
SELECT * FROM productos
WHERE categoria IN ('Electrónica', 'Periféricos');

-- BETWEEN: rango de valores
SELECT * FROM productos
WHERE precio BETWEEN 50 AND 500;

-- LIKE: búsqueda por patrón  (% = cualquier cadena, _ = un carácter)
SELECT * FROM clientes WHERE nombre LIKE 'Ana%';
SELECT * FROM clientes WHERE email LIKE '%@gmail.com';

-- IS NULL / IS NOT NULL
SELECT * FROM clientes WHERE telefono IS NULL;
SELECT * FROM clientes WHERE telefono IS NOT NULL;
PostgreSQL
-- Comparaciones básicas (igual que MySQL)
SELECT * FROM productos WHERE precio > 100;
SELECT * FROM clientes  WHERE activo = TRUE;  -- BOOLEAN real en PG

-- AND / OR / IN / BETWEEN / IS NULL — idénticos a MySQL
SELECT * FROM productos
WHERE categoria IN ('Electrónica', 'Periféricos')
  AND precio BETWEEN 50 AND 500;

-- LIKE (sensible a mayúsculas en PG)
SELECT * FROM clientes WHERE nombre LIKE 'Ana%';

-- ILIKE: igual que LIKE pero sin distinción mayúsculas (solo PG)
SELECT * FROM clientes WHERE nombre ILIKE 'ana%';

-- Operador ~: expresiones regulares (solo PG)
SELECT * FROM clientes WHERE email ~ '^[a-z]+';

-- IS NULL / IS NOT NULL — igual que MySQL
SELECT * FROM clientes WHERE telefono IS NULL;

10 UPDATE — Actualizar datos

Precaución: Siempre usa WHERE en UPDATE y DELETE. Sin WHERE, se modifican o eliminan TODOS los registros de la tabla.
MySQL
-- Actualizar un campo
UPDATE clientes
SET telefono = '555-9999'
WHERE id = 1;

-- Actualizar múltiples campos
UPDATE productos
SET
  precio = 999.99,
  stock  = stock - 1
WHERE id = 3;

-- Aplicar aumento de precio a una categoría
UPDATE productos
SET precio = precio * 1.10
WHERE categoria = 'Electrónica';

-- Ver cuántas filas fueron afectadas
SELECT ROW_COUNT();  -- ejecutar inmediatamente después
PostgreSQL
-- Actualizar un campo
UPDATE clientes
SET telefono = '555-9999'
WHERE id = 1;

-- Actualizar múltiples campos
UPDATE productos
SET
  precio = 999.99,
  stock  = stock - 1
WHERE id = 3;

-- Aplicar aumento de precio
UPDATE productos
SET precio = precio * 1.10
WHERE categoria = 'Electrónica';

-- RETURNING: retornar los registros modificados (exclusivo PG)
UPDATE productos
SET stock = stock - 5
WHERE id = 1
RETURNING id, nombre, stock;

11 DELETE — Eliminar datos

MySQL
-- Eliminar un registro específico
DELETE FROM clientes WHERE id = 5;

-- Eliminar con condición
DELETE FROM productos WHERE stock = 0;

-- Eliminar TODOS los registros (mantiene la tabla)
DELETE FROM clientes;

-- TRUNCATE: elimina todos y reinicia el auto_increment (más rápido)
TRUNCATE TABLE clientes;
PostgreSQL
-- Eliminar un registro específico
DELETE FROM clientes WHERE id = 5;

-- Eliminar con condición
DELETE FROM productos WHERE stock = 0;

-- RETURNING: retornar lo eliminado (exclusivo PG)
DELETE FROM clientes
WHERE activo = FALSE
RETURNING id, nombre, email;

-- TRUNCATE: elimina todos y reinicia la secuencia
TRUNCATE TABLE clientes RESTART IDENTITY;

12 JOINs — Unir tablas

Los JOINs combinan filas de dos o más tablas según una condición de relación.

MySQL — INNER JOIN (solo coincidencias)
-- Pedidos con nombre de cliente y producto
SELECT
  p.id           AS pedido_id,
  c.nombre       AS cliente,
  pr.nombre      AS producto,
  p.cantidad,
  p.total
FROM pedidos p
INNER JOIN clientes  c  ON p.cliente_id  = c.id
INNER JOIN productos pr ON p.producto_id = pr.id;
MySQL — LEFT JOIN (todos los de la izquierda)
-- Todos los clientes, tengan pedidos o no
SELECT
  c.nombre,
  COUNT(p.id) AS total_pedidos
FROM clientes c
LEFT JOIN pedidos p ON c.id = p.cliente_id
GROUP BY c.id, c.nombre
ORDER BY total_pedidos DESC;
MySQL — Clientes SIN pedidos
SELECT c.nombre, c.email
FROM clientes c
LEFT JOIN pedidos p ON c.id = p.cliente_id
WHERE p.id IS NULL;
PostgreSQL — INNER JOIN
-- Idéntico a MySQL en sintaxis
SELECT
  p.id           AS pedido_id,
  c.nombre       AS cliente,
  pr.nombre      AS producto,
  p.cantidad,
  p.total
FROM pedidos p
INNER JOIN clientes  c  ON p.cliente_id  = c.id
INNER JOIN productos pr ON p.producto_id = pr.id;
PostgreSQL — LEFT / RIGHT / FULL OUTER JOIN
-- LEFT JOIN: igual que MySQL
SELECT c.nombre, COUNT(p.id) AS total_pedidos
FROM clientes c
LEFT JOIN pedidos p ON c.id = p.cliente_id
GROUP BY c.id
ORDER BY total_pedidos DESC;

-- FULL OUTER JOIN: registros de ambas tablas aunque no coincidan
SELECT c.nombre, p.id AS pedido
FROM clientes c
FULL OUTER JOIN pedidos p ON c.id = p.cliente_id;

-- CROSS JOIN: producto cartesiano
SELECT c.nombre, pr.nombre
FROM clientes c
CROSS JOIN productos pr;

13 Funciones de agregado

MySQL
-- COUNT: contar registros
SELECT COUNT(*) AS total_clientes FROM clientes;
SELECT COUNT(telefono) AS con_telefono FROM clientes; -- ignora NULLs

-- SUM: suma
SELECT SUM(total) AS ingresos_totales FROM pedidos;

-- AVG: promedio
SELECT AVG(precio) AS precio_promedio FROM productos;

-- MIN / MAX
SELECT
  MIN(precio) AS mas_barato,
  MAX(precio) AS mas_caro
FROM productos;

-- Combinar varias
SELECT
  COUNT(*)         AS total,
  SUM(stock)       AS unidades,
  AVG(precio)      AS precio_avg,
  ROUND(AVG(precio), 2) AS precio_redondeado
FROM productos;
PostgreSQL
-- Las funciones básicas son idénticas a MySQL
SELECT
  COUNT(*)              AS total,
  SUM(stock)            AS unidades,
  ROUND(AVG(precio),2) AS precio_avg,
  MIN(precio)           AS mas_barato,
  MAX(precio)           AS mas_caro
FROM productos;

-- STRING_AGG: concatenar valores de un grupo (solo PG)
SELECT
  categoria,
  STRING_AGG(nombre, ', ' ORDER BY nombre) AS productos
FROM productos
GROUP BY categoria;

-- ARRAY_AGG: agrupar en un arreglo (solo PG)
SELECT
  categoria,
  ARRAY_AGG(nombre) AS lista_productos
FROM productos
GROUP BY categoria;

14 GROUP BY / HAVING

GROUP BY agrupa filas. HAVING filtra grupos (como WHERE pero para agregados).

MySQL
-- Ventas por categoría
SELECT
  categoria,
  COUNT(*) AS cant_productos,
  SUM(stock) AS total_stock,
  ROUND(AVG(precio), 2) AS precio_prom
FROM productos
GROUP BY categoria;

-- HAVING: solo categorías con más de 2 productos
SELECT
  categoria,
  COUNT(*) AS cant
FROM productos
GROUP BY categoria
HAVING COUNT(*) > 2
ORDER BY cant DESC;

-- Pedidos por cliente con HAVING
SELECT
  c.nombre,
  COUNT(p.id)   AS num_pedidos,
  SUM(p.total)  AS total_gastado
FROM clientes c
JOIN pedidos p ON c.id = p.cliente_id
GROUP BY c.id, c.nombre
HAVING SUM(p.total) > 500
ORDER BY total_gastado DESC;
PostgreSQL — igual que MySQL + extras
-- GROUP BY / HAVING — sintaxis idéntica
SELECT
  categoria,
  COUNT(*) AS cant_productos,
  ROUND(AVG(precio),2) AS precio_prom
FROM productos
GROUP BY categoria
HAVING COUNT(*) > 1
ORDER BY precio_prom DESC;

-- ROLLUP: subtotales y gran total (también en MySQL 8+)
SELECT
  categoria,
  SUM(stock) AS total_stock
FROM productos
GROUP BY ROLLUP(categoria);

-- FILTER dentro de agregados (solo PG)
SELECT
  COUNT(*) AS total,
  COUNT(*) FILTER (WHERE precio > 100) AS premium
FROM productos;

15 Subconsultas

MySQL
-- Productos con precio mayor al promedio
SELECT nombre, precio
FROM productos
WHERE precio > (
  SELECT AVG(precio) FROM productos
);

-- Clientes que han realizado al menos un pedido
SELECT nombre, email
FROM clientes
WHERE id IN (
  SELECT DISTINCT cliente_id FROM pedidos
);

-- Subconsulta en FROM (tabla derivada)
SELECT categoria, precio_prom
FROM (
  SELECT categoria, AVG(precio) AS precio_prom
  FROM productos
  GROUP BY categoria
) AS resumen
WHERE precio_prom > 200;
PostgreSQL — subqueries + CTE
-- Subconsultas — misma sintaxis que MySQL
SELECT nombre, precio
FROM productos
WHERE precio > (SELECT AVG(precio) FROM productos);

-- CTE (Common Table Expression) con WITH — más legible
WITH promedio_precios AS (
  SELECT categoria, AVG(precio) AS avg_precio
  FROM productos
  GROUP BY categoria
),
top_categorias AS (
  SELECT * FROM promedio_precios
  WHERE avg_precio > 200
)
SELECT p.nombre, p.precio, t.avg_precio
FROM productos p
JOIN top_categorias t ON p.categoria = t.categoria;

-- EXISTS: más eficiente que IN para grandes volúmenes
SELECT nombre
FROM clientes c
WHERE EXISTS (
  SELECT 1 FROM pedidos p WHERE p.cliente_id = c.id
);

16 Índices

Los índices aceleran las búsquedas en columnas que se consultan frecuentemente.

MySQL
-- Índice simple
CREATE INDEX idx_email ON clientes(email);

-- Índice único
CREATE UNIQUE INDEX idx_email_unico ON clientes(email);

-- Índice compuesto (para queries con múltiples columnas)
CREATE INDEX idx_cat_precio ON productos(categoria, precio);

-- Ver índices de una tabla
SHOW INDEX FROM clientes;

-- Eliminar índice
DROP INDEX idx_email ON clientes;

-- Analizar cómo MySQL usa los índices
EXPLAIN SELECT * FROM clientes WHERE email = 'ana@email.com';
PostgreSQL
-- Índice simple (B-tree por defecto)
CREATE INDEX idx_email ON clientes(email);

-- Índice único
CREATE UNIQUE INDEX idx_email_unico ON clientes(email);

-- Índice compuesto
CREATE INDEX idx_cat_precio ON productos(categoria, precio);

-- Índice parcial: solo indexa productos activos (solo PG)
CREATE INDEX idx_activos ON clientes(email)
WHERE activo = TRUE;

-- Ver índices
\di

-- Eliminar índice
DROP INDEX idx_email;

-- Analizar plan de ejecución
EXPLAIN ANALYZE SELECT * FROM clientes
WHERE email = 'ana@email.com';

17 Vistas (Views)

Una vista es una consulta almacenada que se comporta como una tabla virtual.

MySQL
-- Crear vista: resumen de pedidos por cliente
CREATE VIEW v_resumen_clientes AS
SELECT
  c.id,
  c.nombre,
  c.email,
  COUNT(p.id)  AS total_pedidos,
  SUM(p.total) AS total_gastado
FROM clientes c
LEFT JOIN pedidos p ON c.id = p.cliente_id
GROUP BY c.id, c.nombre, c.email;

-- Consultar la vista como si fuera una tabla
SELECT * FROM v_resumen_clientes

-- Crear vista — misma sintaxis
CREATE VIEW v_resumen_clientes AS
SELECT
  c.id,
  c.nombre,
  c.email,
  COUNT(p.id)  AS total_pedidos,
  SUM(p.total) AS total_gastado
FROM clientes c
LEFT JOIN pedidos p ON c.id = p.cliente_id
GROUP BY c.id;

-- Consultar la vista
SELECT * FROM v_resumen_clientes
ORDER BY total_gastado DESC NULLS LAST;
